Algoritmo "acompanhamento_variaveis"

Var
   valorA, valorB, valorC : inteiro

Inicio
   valorA <- 10
   valorB <- 4
   valorC <- valorA + valorB * 2
   valorA <- valorC - 6
   valorB <- valorA % 5
   valorC <- valorC + valorB - valorA

   escreval("Valores finais das variaveis:")
   escreval("valorA = ", valorA)
   escreval("valorB = ", valorB)
   escreval("valorC = ", valorC)
   escreval("")
   escreva("Resultado final: ", valorA, " ", valorB, " ", valorC)
Fimalgoritmo
